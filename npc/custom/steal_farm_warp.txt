//===== rAthena Script =======================================
//= Steal-Farm Warper
//===== Description: =========================================
//= One flat menu to the six maps plans/steal-farm rotates over, so the
//= headless bot can reach them through routines/lib/warpa.js — which
//= drives a Warper by MENU TEXT — instead of through @warp, which needs
//= group 99 and therefore cannot be part of a path a real account walks.
//=
//= Standalone rather than an edit to npc/custom/warper.txt, for two
//= reasons and the same call session 3 made for steal_farm_healer.txt:
//= that script is duplicate(Warper)'d into ~30 towns, and its field list
//= is GENERATED by `Disp("Prontera Field",0,11); Pick("prt_fild",1)` —
//= a numeric 00..11 range that prt_fild02a/b/c cannot join without
//= rewriting the generator for every town at once.
//=
//= The labels ARE the contract: warpa.js matches exact-then-unique-
//= substring and refuses an ambiguous one, so they are the raw map names
//= the routines already use. `prt_fild02` is a substring of
//= `prt_fild02a`, which is exactly why the exact match outranks the
//= substring one.
//===== Additional Comments: =================================
//= Test server only. The three clones come from
//= roBrowserLegacy-Node/src/Node/tools/map-dup.mjs — identical cells and
//= identical spawns under three names, so a benchmark never waits on
//= another map's 300 s empty-map respawn.
//============================================================

prontera,157,193,4	script	Steal Farm Warper	811,{
	mes "[Steal Farm Warper]";
	mes "Which field?";
	next;
	switch (select("prt_fild02:prt_fild02a:prt_fild02b:prt_fild02c:mjolnir_11:moc_pryd05")) {
		case 1: close2; warp "prt_fild02",240,206; end;
		case 2: close2; warp "prt_fild02a",240,206; end;
		case 3: close2; warp "prt_fild02b",240,206; end;
		case 4: close2; warp "prt_fild02c",240,206; end;
		case 5: close2; warp "mjolnir_11",210,201; end;
		case 6: close2; warp "moc_pryd05",190,8; end;
	}
	end;
}
